cmake_minimum_required (VERSION 3.20)

set(basename "bootstrapper")
project(${basename} C CXX ASM)

onion_module_prologue(C_STANDARD 11 CXX_STANDARD 20)

set(D_CWD "${CMAKE_CURRENT_SOURCE_DIR}")
onion_output_dirs()
set(D_SRC ${D_CWD}/source)

# Cartella di output temporanea per gli asset generati durante la compilazione
set(_icon_output_dir "${CMAKE_CURRENT_BINARY_DIR}/generated/assets")

set(_generated_icon_files)

# ---------------------------------------------------------------------------
# GESTIONE litehen_sicon.png (L'unica icona rimasta)
# ---------------------------------------------------------------------------
set(_raster_icon "litehen_sicon.png")
set(_raster_source "${D_CWD}/assets/${_raster_icon}")
set(_raster_output "${_icon_output_dir}/${_raster_icon}")
add_custom_command(
	OUTPUT "${_raster_output}"
	COMMAND ${CMAKE_COMMAND} -E make_directory "${_icon_output_dir}"
	COMMAND ${CMAKE_COMMAND} -E copy_if_different "${_raster_source}" "${_raster_output}"
	DEPENDS "${_raster_source}"
	COMMENT "Copying ${_raster_icon}"
	VERBATIM
)
list(APPEND _generated_icon_files "${_raster_output}")

# ---------------------------------------------------------------------------
# Generazione dei blocchi di embedding in Assembly e manifest per l'unica icona
# ---------------------------------------------------------------------------
set(_icon_asm "${_icon_output_dir}/icon_assets.inc")
set(_icon_manifest "${_icon_output_dir}/icon_manifest.inc")
set(_asm_body "/* Generated by CMake from bootstrapper/assets — do not edit. */\n")
set(_manifest_body "/* Generated by CMake from bootstrapper/assets — do not edit. */\n")
foreach(_icon_file IN LISTS _generated_icon_files)
	get_filename_component(_icon_name "${_icon_file}" NAME_WE)
	string(APPEND _asm_body
		"\".global ${_icon_name}_start\\n\"\n"
		"\".type   ${_icon_name}_start, @object\\n\"\n"
		"\".align  16\\n\"\n"
		"\"${_icon_name}_start:\\n\"\n"
		"\".incbin \\\"\" LITEHEN _ICON_ASSET_DIR \"/${_icon_name}.png\\\"\\n\"\n"
		"\"${_icon_name}_end:\\n\"\n"
		"\".global ${_icon_name}_size\\n\"\n"
		"\".type   ${_icon_name}_size, @object\\n\"\n"
		"\".align  4\\n\"\n"
		"\"${_icon_name}_size:\\n\"\n"
		"\".int    ${_icon_name}_end - ${_icon_name}_start\\n\"\n")
	string(APPEND _manifest_body "ONION_ICON(${_icon_name})\n")
endforeach()
file(GENERATE OUTPUT "${_icon_asm}" CONTENT "${_asm_body}")
file(GENERATE OUTPUT "${_icon_manifest}" CONTENT "${_manifest_body}")

add_executable(${PROJECT_NAME})
set_target_properties(${PROJECT_NAME} PROPERTIES OUTPUT_NAME "${PROJECT_NAME}.elf")

# Finalize main target sources
target_compile_options(${PROJECT_NAME} PUBLIC
	$<$<COMPILE_LANGUAGE:C>:${C_DEFS} ${C_FLAGS}>
	$<$<COMPILE_LANGUAGE:CXX>:${CXX_DEFS} ${CXX_FLAGS}>
	$<$<COMPILE_LANGUAGE:ASM>:${ASM_FLAGS}>
)

file(GLOB SrcFiles CONFIGURE_DEPENDS ${D_SRC}/*.c ${D_SRC}/*.cpp ${D_SRC}/../include/*.h ${D_SRC}/../include/*.hpp ${D_SRC}/*.s)
list(FILTER SrcFiles EXCLUDE REGEX ".*/(pt|elfldr)\\.c$")

set(CMAKE_C_FLAGS "${ONION_PS5_TARGET_FLAGS} ")
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -Wno-keyword-macro -fPIC -fPIE -march=znver2 -Wall -Werror -fstack-protector-all")
set(CMAKE_CXX_FLAGS "${CMAKE_C_FLAGS} -gfull -gdwarf-2 -pedantic -pedantic-errors")

target_sources(${PROJECT_NAME} PRIVATE ${SrcFiles})

if(DEFINED LITEHEN _OUT_BIN)
	set(_onion_daemon_elf "${LITEHEN _OUT_BIN}/daemon.elf")
	set(_onion_util_elf "${LITEHEN _OUT_BIN}/util.elf")
	set(_onion_elfldr_elf "${LITEHEN _OUT_BIN}/onion_elfldr.elf")
else()
	set(_onion_daemon_elf "${PROJECT_ROOT}/bin/daemon.elf")
	set(_onion_util_elf "${PROJECT_ROOT}/bin/util.elf")
	set(_onion_elfldr_elf "${PROJECT_ROOT}/bin/onion_elfldr.elf")
endif()
set_source_files_properties(
	${D_SRC}/daemon.c
	PROPERTIES OBJECT_DEPENDS
		"${_onion_daemon_elf};${_onion_util_elf};${_onion_elfldr_elf};${LITEHEN _KSTUFF_ELF};${D_CWD}/assets/litehen_sicon.png;${_generated_icon_files}"
)
target_compile_definitions(${PROJECT_NAME} PRIVATE
	LITEHEN _DAEMON_ELF="${_onion_daemon_elf}"
	LITEHEN _UTIL_ELF="${_onion_util_elf}"
	LITEHEN _ELFLDR_ELF="${_onion_elfldr_elf}"
	LITEHEN _KSTUFF_ELF="${LITEHEN _KSTUFF_ELF}"
	LITEHEN _ICON_ASSET_DIR="${_icon_output_dir}"
)

onion_target_debug_define(${PROJECT_NAME})
target_include_directories(${PROJECT_NAME} PRIVATE
	"${D_CWD}/include"
	"${_icon_output_dir}"
	"${PROJECT_ROOT}/libonion_proc/include"
	"${PROJECT_ROOT}/libonion_platform/include"
	"${PROJECT_ROOT}/libonion_settings/include"
	"${PROJECT_ROOT}/libonion_payload/include"
	"${PROJECT_ROOT}/libonion_elfldr/include"
	"${PROJECT_ROOT}/include"
)
target_link_directories	(${PROJECT_NAME} PUBLIC "${LITEHEN _PS5_STUB_DIR}")
target_sources(${PROJECT_NAME} PRIVATE "${LITEHEN _COMMON_DIR}/elfldr_remote.c")
target_link_libraries	(${PROJECT_NAME} PUBLIC onion_proc onion_platform onion_settings onion_payload onion_ready onion_elfldr kernel_sys SceNotification SceSystemService)
add_custom_target(${PROJECT_NAME}_assets
	DEPENDS ${_generated_icon_files} "${_icon_asm}" "${_icon_manifest}")
add_dependencies(${PROJECT_NAME} ${PROJECT_NAME}_assets)
add_dependencies(${PROJECT_NAME} daemon util shellui onion_elfldr_server onion_proc onion_platform onion_settings onion_payload onion_ready onion_elfldr)

# Copy bootstrapper.elf as test.elf before processing
add_custom_command(TARGET ${PROJECT_NAME} POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy $<TARGET_FILE:${PROJECT_NAME}> ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/test.elf
    COMMENT "Copying ${PROJECT_NAME}.elf to test.elf"
)

onion_strip_debug(${PROJECT_NAME})

# ---------------------------------------------------------------------------
# RIPRISTINATO: Blocco di compressione LZMA e creazione target bootstrapper_packed
# ---------------------------------------------------------------------------
set(_bootstrapper_lzma "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/bootstrapper.elf.lzma")
set(_bootstrapper_lzma_size "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/bootstrapper.elf.lzma.size")
add_custom_command(
	OUTPUT "${_bootstrapper_lzma}" "${_bootstrapper_lzma_size}"
	COMMAND ${CMAKE_COMMAND} -E env "TARGET_ELF=$<TARGET_FILE:${PROJECT_NAME}>"
			${PROJECT_ROOT}/../scripts/pack_bootstrapper.sh
	DEPENDS ${PROJECT_NAME}
	COMMENT "Recording size and compressing bootstrapper with LZMA"
)

add_custom_target(${PROJECT_NAME}_packed
	DEPENDS "${_bootstrapper_lzma}" "${_bootstrapper_lzma_size}")
